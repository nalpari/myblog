  4. 해시테이블 직접 만들기

    배열과 해시 함수만으로 Map을 만들어 봅니다. 충돌이 나면 무슨 일이 생기는지, 왜 load factor 0.75에서 두 배로 늘리는지까지.

    class HashTable {
  constructor() { this.buckets = new Array(8).fill(null).map(() => []); this.size = 0; }

  set(key, value) {
    const bucket = this.buckets[hash(key) % this.buckets.length];
    const hit = bucket.find(([k]) => k === key);
    if (hit) hit[1] = value; else { bucket.push([key, value]); this.size++; }
    if (this.size / this.buckets.length > 0.75) this.resize();
  }
}
